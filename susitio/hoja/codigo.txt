/* ---------------- Primera vez ---------------- */
function configurar() {
  const ui = SpreadsheetApp.getUi();
  const r1 = ui.prompt('Susitio', 'Correo con el que entras en Susitio:', ui.ButtonSet.OK_CANCEL);
  if (r1.getSelectedButton() !== ui.Button.OK) return;

  if (r2.getSelectedButton() !== ui.Button.OK) return;
  const props = PropertiesService.getUserProperties();
  props.setProperty('EMAIL', r1.getResponseText().trim());
  props.setProperty('PASS', r2.getResponseText());

  ScriptApp.getProjectTriggers().forEach(t => { if (['actualizar', 'avisos'].includes(t.getHandlerFunction())) ScriptApp.deleteTrigger(t); });

  }

/* ---------------- Leer la nube ---------------- */
function entrar_() {
  const props = PropertiesService.getUserProperties();
  const email = props.getProperty('EMAIL'), pass = props.getProperty('PASS');
  if (!email || !pass) throw new Error('Falta configurar la cuenta: menú Susitio › Cambiar cuenta o contraseña.');
  const r = UrlFetchApp.fetch('https://identitytoolkit.googleapis.com/v1/accounts:signInWithPassword?key=' + API_KEY, {
    method: 'post', contentType: 'application/json', muteHttpExceptions: true,
    payload: JSON.stringify({ email: email, password: pass, returnSecureToken: true })
  });
  const j = JSON.parse(r.getContentText());
  if (!j.idToken) throw new Error('No se ha podido entrar en Susitio: revisa el correo y la contraseña (' + ((j.error && j.error.message) || r.getResponseCode()) + ').');
  return { token: j.idToken, uid: j.localId };
}
function get_(url, token) {
  const r = UrlFetchApp.fetch(url, { headers: { Authorization: 'Bearer ' + token }, muteHttpExceptions: true });
  if (r.getResponseCode() !== 200) throw new Error('La nube ha respondido ' + r.getResponseCode() + ': ' + r.getContentText().slice(0, 200));
  return JSON.parse(r.getContentText());
}
function leerTodo_() {
  const s = entrar_();
  const u = get_(BASE + 'susitioUsuarios/' + s.uid, s.token);
  const casa = u.fields && u.fields.casa && u.fields.casa.stringValue;
  if (!casa) throw new Error('Esta cuenta todavía no está en la casa de Susi. Entra en la app y conéctate primero.');
  const D = { meds: [], tomas: [], semanales: [], cacas: [], comidas: [], apetito: [], cambios: [], diluciones: [], ajustes: [], sintomas: [], energia: [], citas: [], analiticas: [] };
  let page = '', n = 0;
  do {
    const j = get_(BASE + 'susitio/' + casa + '/r?pageSize=300' + (page ? '&pageToken=' + encodeURIComponent(page) : ''), s.token);
    (j.documents || []).forEach(doc => {
      const f = doc.fields || {};
      const col = f.col && f.col.stringValue, d = f.d && f.d.stringValue;
      if (!col || !d || !D[col]) return;
      try { const rec = JSON.parse(d); if (!rec.borrado) { D[col].push(rec); n++; } } catch (e) {}
    });
    page = j.nextPageToken || '';
  } while (page);
  D.total = n;
  return D;
}

/* ---------------- Utilidades ---------------- */
const fecha_ = s => { const p = String(s).split('-').map(Number); return new Date(p[0], p[1] - 1, p[2]); };
const dstr_ = d => d.getFullYear() + '-' + ('0' + (d.getMonth() + 1)).slice(-2) + '-' + ('0' + d.getDate()).slice(-2);
const sumarDias_ = (s, n) => { const d = fecha_(s); d.setDate(d.getDate() + n); return dstr_(d); };
const dow_ = s => (fecha_(s).getDay() + 6) % 7;
const lunes_ = s => sumarDias_(s, -dow_(s));
const media_ = l => l.length ? Math.round(l.reduce((a, b) => a + b, 0) / l.length * 10) / 10 : '';
const horaOrden_ = h => !h ? '0' : (h < '04' ? '2' : '1') + h; // madrugada al final del día
const porFechaDesc_ = (a, b) => (b.fecha || '').localeCompare(a.fecha || '') || horaOrden_(b.hora).localeCompare(horaOrden_(a.hora));
const nombreCambio_ = c => c.tipo === 'dieta' && c.alimento ? 'Solo ' + c.alimento : (c.texto || TIPO_CAMBIO[c.tipo] || '');
function pauta_(m) {
  if (m.tipo === 'diario') { const mm = m.momentos || []; return mm.length === 2 ? 'Cada día · mañana y noche' : mm.length ? 'Cada día · ' + MOMENTO[mm[0]].toLowerCase() : 'Cada día · fuera de las tomas'; }
  const d = m.dias || []; return d.length ? 'Cada semana · ' + d.map(x => DIAS_L[x]).join(' y ') : 'Una vez a la semana';
}

function escribir_(ss, nombre, cabecera, filas, formatos) {
  let sh = ss.getSheetByName(nombre);
  if (!sh) sh = ss.insertSheet(nombre);
  sh.clearContents();
  const datos = [cabecera].concat(filas);
  sh.getRange(1, 1, datos.length, cabecera.length).setValues(datos.map(r => cabecera.map((_, i) => r[i] === undefined || r[i] === null ? '' : r[i])));
  const cab = sh.getRange(1, 1, 1, cabecera.length);
  cab.setFontWeight('bold').setBackground('#1D6B66').setFontColor('#FFFFFF');
  sh.setFrozenRows(1);
  if (filas.length && formatos) Object.keys(formatos).forEach(col => sh.getRange(2, Number(col) + 1, filas.length, 1).setNumberFormat(formatos[col]));
  // quitar filas sobrantes de una actualización anterior más larga
  const sobra = sh.getMaxRows() - datos.length;
  if (sobra > 50) sh.deleteRows(datos.length + 1, sobra - 50);
  sh.autoResizeColumns(1, cabecera.length);
  return sh;
}

/* ================= Avisos al móvil =================
 * Cada 5 minutos mira cuál fue la última toma. A las 11 h 30 min manda un
 * aviso a los móviles que lo hayan activado en la app, si la siguiente toma
 * todavía no está apuntada. Solo avisa una vez por toma.
 */
function tomaMs_(t) { const d = fecha_(t.fecha); const h = (t.hora || '00:00').split(':').map(Number); d.setHours(h[0], h[1], 0, 0); if ((t.hora || '') < '04') d.setDate(d.getDate() + 1); return d.getTime(); }
function casa_(s) { const u = get_(BASE + 'susitioUsuarios/' + s.uid, s.token); const c = u.fields && u.fields.casa && u.fields.casa.stringValue; if (!c) throw new Error('Esta cuenta no está en la casa de Susi.'); return c; }
function recientes_(s, casa, desdeMs) {
  // registros cambiados en los últimos días
  const r = UrlFetchApp.fetch(BASE + 'susitio/' + casa + ':runQuery', {
    method: 'post', contentType: 'application/json', muteHttpExceptions: true, headers: { Authorization: 'Bearer ' + s.token },
    payload: JSON.stringify({ structuredQuery: { from: [{ collectionId: 'r' }], where: { fieldFilter: { field: { fieldPath: 'upd' }, op: 'GREATER_THAN', value: { integerValue: String(desdeMs) } } } } })
  });
  if (r.getResponseCode() !== 200) throw new Error('La nube ha respondido ' + r.getResponseCode() + ': ' + r.getContentText().slice(0, 200));
  const out = { tomas: [], meds: [] };
  JSON.parse(r.getContentText()).forEach(x => {
    const f = x.document && x.document.fields; if (!f || !f.col || !f.d) return;
    const col = f.col.stringValue; if (!out[col]) return;
    try { const rec = JSON.parse(f.d.stringValue); if (!rec.borrado) out[col].push(rec); } catch (e) {}
  });
  return out;
}
function todosMeds_(s, casa) {
  const r = UrlFetchApp.fetch(BASE + 'susitio/' + casa + ':runQuery', {
    method: 'post', contentType: 'application/json', muteHttpExceptions: true, headers: { Authorization: 'Bearer ' + s.token },
    payload: JSON.stringify({ structuredQuery: { from: [{ collectionId: 'r' }], where: { fieldFilter: { field: { fieldPath: 'col' }, op: 'EQUAL', value: { stringValue: 'meds' } } } } })
  });
  const l = [];
  try { JSON.parse(r.getContentText()).forEach(x => { const f = x.document && x.document.fields; if (f && f.d) { const m = JSON.parse(f.d.stringValue); if (!m.borrado) l.push(m); } }); } catch (e) {}
  return l;
}
// Dos avisos por toma: a las 11 h 30 min ("quedan 30 min") y a las 11 h 55 min ("hora de preparar").
// Cada 5 minutos se mira la última toma y se programan los dos avisos a su hora exacta.
const ETAPAS = { a30: 11 * 60 + 30, a55: 11 * 60 + 55 };
function siguiente_(s, casa) {
  const R = recientes_(s, casa, Date.now() - 3 * 24 * 3600000);
  const u = R.tomas.filter(t => t.hora).sort((a, b) => tomaMs_(b) - tomaMs_(a))[0];
  return u ? { u: u, clave: u.id + '@' + u.hora } : null;
}
function avisos() {
  const s = entrar_(), casa = casa_(s), x = siguiente_(s, casa);
  if (!x) return;
  const props = PropertiesService.getScriptProperties(), base = tomaMs_(x.u), ahora = Date.now();
  if (props.getProperty('AVISO_TOMA') !== x.clave) {
    // toma nueva: se programan los dos avisos a su hora
    ScriptApp.getProjectTriggers().forEach(t => { if (['aviso30', 'aviso55'].includes(t.getHandlerFunction())) ScriptApp.deleteTrigger(t); });
    if (base + ETAPAS.a30 * 60000 > ahora + 60000) ScriptApp.newTrigger('aviso30').timeBased().at(new Date(base + ETAPAS.a30 * 60000)).create();
    if (base + ETAPAS.a55 * 60000 > ahora + 60000) ScriptApp.newTrigger('aviso55').timeBased().at(new Date(base + ETAPAS.a55 * 60000)).create();
    props.setProperties({ AVISO_TOMA: x.clave, ENVIADO_a30: '', ENVIADO_a55: '' });
  }
  // por si un aviso programado no saltó: se manda aquí, con unos minutos de retraso
  const t55 = base + ETAPAS.a55 * 60000;
  if (ahora >= t55 && ahora < t55 + 3 * 3600000) mandarEtapa_('a55', s, casa, x);
  else if (ahora >= base + ETAPAS.a30 * 60000 && ahora < t55) mandarEtapa_('a30', s, casa, x);
}
function aviso30() { mandarEtapa_('a30'); }
function aviso55() { mandarEtapa_('a55'); }
function mandarEtapa_(etapa, s, casa, x) {
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(20000)) return;
  try {
    s = s || entrar_(); casa = casa || casa_(s); x = x || siguiente_(s, casa);
    if (!x) return;
    const props = PropertiesService.getScriptProperties();
    if (props.getProperty('ENVIADO_' + etapa) === x.clave) return; // ya enviado
    const t = tomaMs_(x.u) + ETAPAS[etapa] * 60000, ahora = Date.now();
    if (ahora < t - 2 * 60000 || ahora > t + 3 * 3600000) return; // no toca ahora (o ya se dio la siguiente toma)
    const sig = x.u.momento === 'manana' ? 'noche' : 'manana';
    const hora = Utilities.formatDate(new Date(tomaMs_(x.u) + 12 * 3600000), Session.getScriptTimeZone(), 'HH:mm');
    const meds = todosMeds_(s, casa).filter(m => m.activo && m.tipo === 'diario' && (m.momentos || []).indexOf(sig) >= 0).map(m => m.nombre + ' ' + (m.dosis || ''));
    const lista = meds.length ? meds.join(', ') : 'la toma';
    const nom = sig === 'noche' ? 'noche' : 'mañana';
    const n = etapa === 'a30'
      ? enviar_(s, casa, 'Quedan 30 min · toma de la ' + nom + ' (' + hora + ')', 'Susi: ' + lista + '.')
      : enviar_(s, casa, 'Hora de preparar la medicina', 'Toma de la ' + nom + ' de Susi a las ' + hora + ': ' + lista + '. No olvides pesarla.');
    if (n) props.setProperty('ENVIADO_' + etapa, x.clave);
  } finally { lock.releaseLock(); }
}
function activarAvisos() {
  ScriptApp.getProjectTriggers().forEach(t => { if (t.getHandlerFunction() === 'avisos') ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('avisos').timeBased().everyMinutes(5).create();
  avisoPrueba();
}
function avisoPrueba() {
  const s = entrar_(), casa = casa_(s);
  const n = enviar_(s, casa, 'Prueba de Susitio', 'Si ves esto, los avisos funcionan 🐾');
  SpreadsheetApp.getUi().alert('Aviso de prueba', n ? 'Enviado a ' + n + ' móvil(es).' : 'No hay ningún móvil con los avisos activados. Actívalos en la app: Medicación › Nube › Activar avisos.', SpreadsheetApp.getUi().ButtonSet.OK);
}
function enviar_(s, casa, titulo, texto) {
  const j = get_(BASE + 'susitio/' + casa + '/avisos?pageSize=50', s.token);
  const docs = j.documents || [];
  let n = 0;
  const oauth = ScriptApp.getOAuthToken();
  docs.forEach(d => {
    const tok = d.fields && d.fields.token && d.fields.token.stringValue;
    if (!tok) return;
    const r = UrlFetchApp.fetch('https://fcm.googleapis.com/v1/projects/' + PROYECTO + '/messages:send', {
      method: 'post', contentType: 'application/json', muteHttpExceptions: true,
      headers: { Authorization: 'Bearer ' + oauth, 'x-goog-user-project': PROYECTO },
      payload: JSON.stringify({ message: { token: tok, data: { title: titulo, body: texto, tag: 'toma' }, webpush: { headers: { Urgency: 'high', TTL: '10800' } } } })
    });
    const c = r.getResponseCode();
    const quien = d.fields.nombre ? d.fields.nombre.stringValue : '?';
    console.log('Aviso «' + titulo + '» → ' + quien + ': ' + c + (c === 200 ? '' : ' ' + r.getContentText().slice(0, 300)));
    if (c === 200) n++;
    else if (c === 404 || /UNREGISTERED/.test(r.getContentText())) {
      // ese móvil ya no admite avisos: se quita
      UrlFetchApp.fetch('https://firestore.googleapis.com/v1/' + d.name, { method: 'delete', headers: { Authorization: 'Bearer ' + s.token }, muteHttpExceptions: true });
    } else console.log('Aviso no enviado (' + c + '): ' + r.getContentText().slice(0, 300));
  });
  return n;
}
